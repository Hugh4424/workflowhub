import { execFileSync } from "node:child_process";
import { mkdtempSync, mkdirSync, realpathSync, rmSync, writeFileSync } from "node:fs";
import { tmpdir } from "node:os";
import { join } from "node:path";
import { afterEach, describe, expect, it } from "vitest";

import { createTask } from "../../runtime/task/task-handle.mjs";
import { prepareTaskWorkspace } from "../../runtime/task/workspace.mjs";
import { assertCurrentSourceDigest, ensureGitSnapshotObjectStore } from "../../runtime/task/git-worktree-snapshot.mjs";

const roots = [];
const git = (cwd, ...args) => execFileSync("git", args, { cwd, encoding: "utf8", stdio: ["ignore", "pipe", "pipe"] }).trim();

function taskFixture(mode) {
  const root = realpathSync(mkdtempSync(join(tmpdir(), "workflowhub-p9-scope-")));
  roots.push(root);
  const repo = join(root, "repo");
  mkdirSync(repo);
  git(repo, "init", "-q", "-b", "main");
  git(repo, "config", "user.name", "Scope fixture");
  git(repo, "config", "user.email", "scope@example.test");
  writeFileSync(join(repo, "product.mjs"), "export const version = 0;\n");
  writeFileSync(join(repo, "old-name.mjs"), "export const oldName = true;\n");
  writeFileSync(join(repo, "to-delete.mjs"), "export const obsolete = true;\n");
  git(repo, "add", ".");
  git(repo, "commit", "-qm", "before task starts");
  const startCommit = git(repo, "rev-parse", "HEAD");
  const worktree = join(root, "task-worktree");
  mkdirSync(join(root, "storage"));
  if (mode === "existing") git(repo, "worktree", "add", "-q", "-b", "task/p9-existing", worktree, "HEAD");
  const task = createTask({
    storageRoot: join(root, "storage"),
    manifest: {
      schema_version: "1.0.0", project_name: "Scope", task_id: mode === "existing" ? "p9-existing" : "p9-created",
      created_at: "2026-09-19T00:00:00Z", target_repo_root: repo, issue_ids: [], inputs: {},
      record_model: "vnext-single-write", activation_cohort: "post",
      ...(mode === "existing" ? { workspace_mode: "existing", workspace_root: worktree } : {}),
    },
  });
  return { repo, root, task, startCommit, worktree: mode === "existing" ? worktree : null };
}

afterEach(() => {
  for (const root of roots.splice(0)) rmSync(root, { recursive: true, force: true });
});

describe("ORACLE-P9-TRUSTED-CHANGE-SCOPE — task-start baseline and repair", () => {
  it.each(["existing", "created"])("retains the certified task-start commit across a clean committed %s workspace", (mode) => {
    const state = taskFixture(mode);
    const first = prepareTaskWorkspace(state.task);
    const worktree = first.worktreeRoot;
    expect(first.baselineCommit).toBe(state.startCommit);
    writeFileSync(join(worktree, "product.mjs"), "export const version = 1;\n");
    git(worktree, "add", "product.mjs");
    git(worktree, "commit", "-qm", "task implementation");
    expect(git(worktree, "status", "--porcelain")).toBe("");
    const current = prepareTaskWorkspace(state.task);
    const snapshot = current.captureSnapshot();
    ensureGitSnapshotObjectStore(worktree);
    const actualChanged = git(worktree, "diff", "--name-only", current.baselineCommit, snapshot.commit).split("\n").filter(Boolean);
    expect(current.baselineCommit, "certified task-start commit must not silently become current HEAD").toBe(state.startCommit);
    expect(actualChanged, "a clean committed implementation still changes product.mjs since task start").toContain("product.mjs");
  });

  it("captures staged, unstaged, untracked, renamed and deleted source bytes after repair", () => {
    const state = taskFixture("created");
    const candidate = prepareTaskWorkspace(state.task);
    const worktree = candidate.worktreeRoot;
    writeFileSync(join(worktree, "product.mjs"), "export const version = 1;\n");
    git(worktree, "add", "product.mjs");
    git(worktree, "commit", "-qm", "first repair");
    git(worktree, "mv", "old-name.mjs", "renamed.mjs");
    git(worktree, "rm", "-q", "to-delete.mjs");
    writeFileSync(join(worktree, "staged.mjs"), "export const staged = true;\n");
    git(worktree, "add", "staged.mjs");
    writeFileSync(join(worktree, "product.mjs"), "export const version = 2;\n");
    writeFileSync(join(worktree, "untracked.mjs"), "export const untracked = true;\n");
    const snapshot = candidate.captureSnapshot();
    ensureGitSnapshotObjectStore(worktree);
    const changed = git(worktree, "diff", "--no-renames", "--name-only", candidate.baselineCommit, snapshot.commit).split("\n").filter(Boolean);
    expect(changed).toEqual(expect.arrayContaining([
      "product.mjs", "old-name.mjs", "renamed.mjs", "to-delete.mjs", "staged.mjs", "untracked.mjs",
    ]));
    expect(snapshot.source_manifest.entries.find(({ path }) => path === "product.mjs")?.status).toBe("present");
    expect(changed).toContain("to-delete.mjs");
    expect(snapshot.source_manifest.entries.find(({ path }) => path === "untracked.mjs")?.status).toBe("present");
    writeFileSync(join(worktree, "product.mjs"), "export const version = 3;\n");
    expect(() => assertCurrentSourceDigest(worktree, snapshot.source_digest, state.task.identity.taskId, "post"))
      .toThrow(/FORMAL_SNAPSHOT_MISMATCH/);
  });
});
